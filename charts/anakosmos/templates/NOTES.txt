{{- if .Values.ingress.enabled }}
🌐 Access the dashboard at:
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
🌐 Access the dashboard at:
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "anakosmos.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
🌐 Access the dashboard at:
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        You can watch the status by running:
        kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "anakosmos.fullname" . }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "anakosmos.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
🌐 Access the dashboard by port-forwarding:
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "anakosmos.fullname" . }} 8080:{{ .Values.service.port }}
  echo "Visit http://localhost:8080"
{{- end }}
{{- if .Values.inClusterMode.enabled }}

⚠️  WARNING: In-cluster mode is ENABLED.
   The dashboard will auto-connect using its ServiceAccount WITHOUT user authentication.
   Anyone who can reach the dashboard can administer the cluster.
   Only enable this in trusted networks or behind an authenticating proxy.
{{- else }}

🔒 In-cluster mode is disabled (default). The internal proxy is blocked.
   Users must connect to a remote cluster and provide their own credentials.
{{- end }}
{{- if .Values.serviceAccount.create }}

📋 Retrieve the ServiceAccount token with:

  kubectl create token {{ include "anakosmos.serviceAccountName" . }} --namespace {{ .Release.Namespace }}

   Or, for a long-lived token, create a Secret:

  kubectl apply -f - <<EOF
  apiVersion: v1
  kind: Secret
  metadata:
    name: {{ include "anakosmos.serviceAccountName" . }}-token
    namespace: {{ .Release.Namespace }}
    annotations:
      kubernetes.io/service-account.name: {{ include "anakosmos.serviceAccountName" . }}
  type: kubernetes.io/service-account-token
  EOF

  kubectl get secret {{ include "anakosmos.serviceAccountName" . }}-token --namespace {{ .Release.Namespace }} -o jsonpath='{.data.token}' | base64 -d
{{- end }}
